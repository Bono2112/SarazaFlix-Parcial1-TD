    class Usuario {
        <<entidad>>
        +string Id
        +string Nombre
        +Plan Plan
    }

    class Contenido {
        <<entidad>>
        +string Id
        +string Titulo
        +Plan PlanMinimo
        +EstaIncluidoEn(Plan) bool
    }

    class Reproduccion {
        <<registro que se persiste>>
        +Usuario Usuario
        +Contenido Contenido
        +IReadOnlyList~CalidadReproduccion~ CalidadesUsadas
        +bool ServidoDesdeDispositivo
        +List~string~ IntentosRechazados
        +RegistrarCalidad(CalidadReproduccion) void
    }

    class Plan {
        <<enumeration>>
        Basico
        Estandar
        Premium
    }

    class CalidadConexion {
        <<enumeration>>
        Mala
        Media
        Buena
    }

    class CalidadReproduccion {
        <<enumeration>>
        Baja
        HD
        CuatroK
    }

    class IFuenteContenido {
        <<interface>>
        +Obtener(string idContenido) Contenido
    }

    class ServidorRemotoContenido {
        <<servidor remoto>>
        +IReadOnlyCollection~Contenido~ Catalogo
        +int Descargas
        +Obtener(string idContenido) Contenido
    }

    class IRepositorioReproduccion {
        <<interface - Repository>>
        +Guardar(Reproduccion) void
        +ObtenerTodas() IReadOnlyList~Reproduccion~
    }

    class RepositorioReproduccionArchivo {
        <<Repository concreto>>
        +string Ruta
        formato de texto personalizado (;)
    }

    class IEstadoReproduccion {
        <<interface - State>>
        +CalidadReproduccion Calidad
        +Reproducir(Contenido) string
        +Siguiente(CalidadConexion) IEstadoReproduccion
    }

    class ConexionBuena {
        <<estado concreto: conexion buena, reproduce en 4K>>
    }

    class ConexionMedia {
        <<estado concreto: conexion media, reproduce en HD>>
    }

    class ConexionMala {
        <<estado concreto: conexion mala, reproduce en baja>>
    }

    class AhorroDeDatos {
        <<estado concreto: modo ahorro, siempre en baja>>
        +IEstadoReproduccion EstadoDeConexion
    }

    class FabricaEstado {
        <<Factory Method - tabla de transiciones>>
        +Crear(CalidadConexion) IEstadoReproduccion
    }

    class IObservadorConexion {
        <<interface - Observer>>
        +AlCambiarConexion(CalidadConexion) void
    }

    class MonitorConexion {
        <<Observer - subject>>
        +CalidadConexion UltimaMedicion
        +Suscribir(IObservadorConexion) void
        +Desuscribir(IObservadorConexion) void
        +Medir(CalidadConexion) void
    }

    class ProxyControlPlan {
        <<Proxy - control de acceso>>
        +List~string~ IntentosRechazados
        +Obtener(string idContenido) Contenido
    }

    class ProxyCacheLocal {
        <<Proxy - cache del dispositivo>>
        +IReadOnlyCollection~Contenido~ ContenidosEnDispositivo
        +bool UltimoSirvioDesdeDispositivo
        +Obtener(string idContenido) Contenido
    }

    class ExportadorBase~T~ {
        <<Template Method + Reflexion>>
        #string Separador
        +Exportar(IEnumerable~T~) string
        +Exportar(string ruta, IEnumerable~T~) void
    }

    class ExportadorTexto~T~ {
        <<formato concreto>>
    }

    class Reproductor {
        <<contexto de State / Observer>>
        -IEstadoReproduccion estado
        +Reproduccion Registro
        +CalidadConexion Conexion
        +bool ModoAhorro
        +CalidadReproduccion CalidadActual
        +Reproducir() string
        +AlCambiarConexion(CalidadConexion) void
        +ActivarModoAhorro(bool) void
        +CambiarContenido(string) bool
        -CambiarEstado(IEstadoReproduccion) void
    }

    class ServicioReproduccion {
        <<arma la cadena de proxies>>
        +Iniciar(Usuario, string idContenido) Reproductor
        +Detener(Reproductor) void
    }

    class Consola {
        <<Presentacion>>
        +Main() void
    }
